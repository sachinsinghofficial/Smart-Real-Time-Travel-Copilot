cmake_minimum_required(VERSION 3.24)
project(travelsetu_compute LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(travelsetu-compute engine.cpp)
target_link_libraries(travelsetu-compute PRIVATE Threads::Threads)
option(TRAVELSETU_CUDA "Build optional CUDA dot product kernel" OFF)
if(TRAVELSETU_CUDA)
  enable_language(CUDA)
  target_sources(travelsetu-compute PRIVATE accel.cu)
  target_compile_definitions(travelsetu-compute PRIVATE TRAVELSETU_CUDA=1)
  set_target_properties(travelsetu-compute PROPERTIES CUDA_STANDARD 17)
endif()
option(TRAVELSETU_POSTGRES "Build operator libpqxx report tool" OFF)
if(TRAVELSETU_POSTGRES)
  find_package(libpqxx CONFIG REQUIRED)
  add_executable(travelsetu-report analytics.cpp)
  target_link_libraries(travelsetu-report PRIVATE pqxx)
endif()
enable_testing()
add_test(NAME math COMMAND travelsetu-compute --self-test)
